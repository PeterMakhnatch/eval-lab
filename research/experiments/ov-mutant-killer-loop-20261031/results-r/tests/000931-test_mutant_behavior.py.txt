import io
import sys
import unittest

sys.path.insert(0, '/testbed')

import fastavro
from fastavro import _read_py


def _write(schema, records):
    bio = io.BytesIO()
    fastavro.writer(bio, schema, records)
    bio.seek(0)
    return bio


def _read_both(bio, reader_schema):
    # Pure-python reader (where the dict/complex matching lives) and public one
    data = bio.getvalue()
    py = list(_read_py.reader(io.BytesIO(data), reader_schema))
    pub = list(fastavro.reader(io.BytesIO(data), reader_schema))
    return py, pub


class TestComplexTypeSchemaMatching(unittest.TestCase):
    def test_array_of_evolved_records(self):
        writer = {
            "type": "record",
            "name": "Top",
            "fields": [{
                "name": "items",
                "type": {
                    "type": "array",
                    "items": {
                        "type": "record",
                        "name": "Item",
                        "fields": [{"name": "a", "type": "string"}],
                    },
                },
            }],
        }
        reader = {
            "type": "record",
            "name": "Top",
            "fields": [{
                "name": "items",
                "type": {
                    "type": "array",
                    "items": {
                        "type": "record",
                        "name": "Item",
                        "fields": [
                            {"name": "a", "type": "string"},
                            {"name": "b", "type": "int", "default": 7},
                        ],
                    },
                },
            }],
        }
        bio = _write(writer, [{"items": [{"a": "x"}, {"a": "y"}]}])
        expected = [{"items": [{"a": "x", "b": 7}, {"a": "y", "b": 7}]}]
        py, pub = _read_both(bio, reader)
        self.assertEqual(py, expected)
        self.assertEqual(pub, expected)

    def test_map_of_evolved_records(self):
        writer = {
            "type": "record",
            "name": "Top",
            "fields": [{
                "name": "m",
                "type": {
                    "type": "map",
                    "values": {
                        "type": "record",
                        "name": "Val",
                        "fields": [{"name": "a", "type": "string"}],
                    },
                },
            }],
        }
        reader = {
            "type": "record",
            "name": "Top",
            "fields": [{
                "name": "m",
                "type": {
                    "type": "map",
                    "values": {
                        "type": "record",
                        "name": "Val",
                        "fields": [
                            {"name": "a", "type": "string"},
                            {"name": "c", "type": ["null", "string"],
                             "default": None},
                        ],
                    },
                },
            }],
        }
        bio = _write(writer, [{"m": {"k": {"a": "v"}}}])
        expected = [{"m": {"k": {"a": "v", "c": None}}}]
        py, pub = _read_both(bio, reader)
        self.assertEqual(py, expected)
        self.assertEqual(pub, expected)

    def test_array_of_records_with_nested_array_union_added(self):
        writer = {
            "type": "record",
            "name": "D",
            "fields": [{
                "name": "sub",
                "type": {
                    "type": "array",
                    "items": {
                        "type": "record",
                        "name": "S",
                        "fields": [{"name": "s", "type": "string"}],
                    },
                },
            }],
        }
        reader = {
            "type": "record",
            "name": "D",
            "fields": [{
                "name": "sub",
                "type": {
                    "type": "array",
                    "items": {
                        "type": "record",
                        "name": "S",
                        "fields": [
                            {"name": "s", "type": "string"},
                            {"name": "logs", "default": None, "type": [
                                "null",
                                {"type": "array", "items": "string"},
                            ]},
                        ],
                    },
                },
            }],
        }
        bio = _write(writer, [{"sub": [{"s": "foo"}]}])
        expected = [{"sub": [{"s": "foo", "logs": None}]}]
        py, pub = _read_both(bio, reader)
        self.assertEqual(py, expected)
        self.assertEqual(pub, expected)


if __name__ == '__main__':
    unittest.main()
