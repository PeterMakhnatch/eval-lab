import os
import sys

os.environ.setdefault("MPLBACKEND", "Agg")
sys.path.insert(0, "/testbed")

import unittest
import numpy as np

from impedance.models.circuits import (
    CustomCircuit,
    calculateCircuitLength,
    check_and_eval,
)
from impedance.models.circuits.elements import RR, circuit_elements


class TestMultiLetterContract(unittest.TestCase):
    def test_rr_element_metadata_and_lookup(self):
        self.assertEqual(RR.num_params, 1)
        self.assertEqual(RR.units, ["Ohm"])
        self.assertIn("RR", circuit_elements)
        self.assertIs(circuit_elements["RR"], RR)

    def test_check_and_eval_validation(self):
        self.assertIs(check_and_eval("RR"), RR)
        for invalid in ["rr", "r", "c", "RA", "RZ", "RRX", "R2D", "R0", "1"]:
            with self.assertRaises(ValueError):
                check_and_eval(invalid)

    def test_calculate_circuit_length(self):
        self.assertEqual(calculateCircuitLength(""), 0)
        self.assertEqual(calculateCircuitLength("RR0-R1"), 2)
        self.assertEqual(calculateCircuitLength("RR0-RR1"), 2)
        self.assertEqual(calculateCircuitLength("RR0-W1"), 3)
        self.assertEqual(
            calculateCircuitLength("R0-p(RR0,C1)-p(R1,C2032478)-W1"), 7
        )

    def test_public_param_names_and_indices_preservation(self):
        model1 = CustomCircuit(
            "R0-p(RR0,C1)-p(R1,C2032478)-W1",
            initial_guess=[1.0, 0.01, 2.0, 0.02, 3.0, 4.0, 5.0],
        )
        names1, units1 = model1.get_param_names()
        self.assertEqual(
            names1, ["R0", "RR0", "C1", "R1", "C2032478", "W1_0", "W1_1"]
        )
        self.assertEqual(
            units1, ["Ohm", "Ohm", "F", "Ohm", "F", "Ohm", "sec"]
        )

        model2 = CustomCircuit(
            "RR12345678-p(C34567890,R987654321)-L10000000",
            initial_guess=[0.1, 0.01, 3.0, 0.001],
        )
        names2, _ = model2.get_param_names()
        self.assertEqual(
            names2, ["RR12345678", "C34567890", "R987654321", "L10000000"]
        )

    def test_invalid_circuit_element_names_rejected(self):
        frequencies = np.array([1.0, 10.0])
        for circuit in (
            "Aa0-R1",
            "R2D0-R1",
            "Ra0-R1",
            "RR2D0-R1",
            "rr0-R1",
            "RRA0-R1",
            "RZ0-R1",
            "RRX12-R1",
            "RA0-R1",
        ):
            with self.assertRaises(Exception):
                m = CustomCircuit(circuit, initial_guess=[1.0, 1.0])
                m.predict(frequencies, use_initial=True)

    def test_rr_scaling_factor_exact_100(self):
        frequencies = np.array([0.5, 2.0, 8.0, 32.0])
        model = CustomCircuit("RR0-R1", initial_guess=[2.5, 1.0])
        predicted = model.predict(frequencies, use_initial=True)
        expected = np.array([251.0, 251.0, 251.0, 251.0])
        np.testing.assert_allclose(predicted, expected)

        r_model = CustomCircuit("R0-R1", initial_guess=[250.0, 1.0])
        np.testing.assert_allclose(
            predicted, r_model.predict(frequencies, use_initial=True)
        )

    def test_fit_with_constants_bounds_alignment(self):
        frequencies = np.logspace(-1, 4, 32)
        r_const = 7.0
        rr_true = 0.23
        target_z = np.full(frequencies.shape, 100.0 * rr_true + r_const + 0j)
        model = CustomCircuit(
            circuit="RR0-R1", initial_guess=[0.11], constants={"R1": r_const}
        )
        model.fit(frequencies, target_z)
        np.testing.assert_allclose(
            model.parameters_, [rr_true], rtol=1e-4, atol=1e-7
        )

    def test_fit_warburg_non_cpe_upper_bound_above_one(self):
        frequencies = np.logspace(0, 3, 20)
        true_params = [10.0, 50.0, 5.0]
        model = CustomCircuit("R0-W1", initial_guess=true_params)
        target = model.predict(frequencies, use_initial=True)
        fit_model = CustomCircuit("R0-W1", initial_guess=[10.0, 50.0, 5.0])
        fit_model.fit(frequencies, target)
        np.testing.assert_allclose(
            fit_model.parameters_, true_params, rtol=1e-3, atol=1e-5
        )
        self.assertGreater(fit_model.parameters_[2], 1.0)

    def test_multiletter_multidigit_optimization(self):
        frequencies = np.logspace(-1, 5, 48)
        rr_value = 0.37
        capacitance = 2.3e-4
        impedance = 100.0 * rr_value + 1.0 / (
            2.0j * np.pi * frequencies * capacitance
        )
        model = CustomCircuit(circuit="RR12-C345", initial_guess=[0.31, 2.0e-4])
        model.fit(frequencies, impedance)
        np.testing.assert_allclose(
            model.parameters_, [rr_value, capacitance], rtol=5e-3, atol=1e-8
        )


if __name__ == "__main__":
    unittest.main()
