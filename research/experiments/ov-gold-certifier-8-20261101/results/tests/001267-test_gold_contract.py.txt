import json
import sys
import unittest

sys.path.insert(0, "/testbed/tests")
sys.path.insert(0, "/testbed")

from flask import url_for
from redash import models
from tests.test_controllers import BaseTestCase, app, authenticated_user, query_factory


class TestQuerySearchGoldContract(BaseTestCase):
    def test_endpoint_registration_and_authentication(self):
        self.assertIn("queries_search", app.view_functions)
        with app.test_request_context():
            self.assertEqual(url_for("queries_search"), "/api/queries/search")
            self.assertEqual(
                url_for("queries_search", q="findme"),
                "/api/queries/search?q=findme",
            )

        with app.test_client() as client:
            resp = client.get("/api/queries/search?q=findme")
            self.assertNotEqual(resp.status_code, 200)
            self.assertIn(resp.status_code, (302, 401, 403))

    def test_http_search_parameters_and_serialization(self):
        target = query_factory.create(
            name="AlphaSearchToken report",
            description="Details for AlphaSearchToken",
        )
        with app.test_client() as client, authenticated_user(client):
            # Normal search with serialized dictionary response
            resp = client.get(
                "/api/queries/search", query_string={"q": "alphasearchtoken"}
            )
            self.assertEqual(resp.status_code, 200)
            self.assertTrue(resp.content_type.startswith("application/json"))
            items = json.loads(resp.data)
            self.assertIsInstance(items, list)
            matched = [it for it in items if it["id"] == target.id]
            self.assertEqual(len(matched), 1)
            self.assertEqual(matched[0]["name"], target.name)
            self.assertEqual(matched[0]["description"], target.description)

            # Unmatched term returns empty list
            resp_empty = client.get(
                "/api/queries/search",
                query_string={"q": "nonexistent_term_xyz999"},
            )
            self.assertEqual(json.loads(resp_empty.data), [])

            # Missing q parameter defaults to empty string
            resp_no_q = client.get("/api/queries/search")
            self.assertEqual(resp_no_q.status_code, 200)
            ids_no_q = [it["id"] for it in json.loads(resp_no_q.data)]
            self.assertIn(target.id, ids_no_q)

            # Explicit empty q parameter returns queries
            resp_blank = client.get(
                "/api/queries/search", query_string={"q": ""}
            )
            self.assertEqual(resp_blank.status_code, 200)
            ids_blank = [it["id"] for it in json.loads(resp_blank.data)]
            self.assertIn(target.id, ids_blank)

    def test_http_search_matching_and_or_semantics(self):
        name_only = query_factory.create(
            name="BetaRaven metrics query",
            description="Ordinary summary",
        )
        desc_only = query_factory.create(
            name="Ordinary summary title",
            description="Contains BetaRaven metrics details",
        )
        sql_only = query_factory.create(
            name="Ordinary sql title",
            description="Ordinary sql desc",
            query="SELECT * FROM BetaRaven",
        )
        neither = query_factory.create(
            name="Completely different title",
            description="Completely different desc",
        )
        with app.test_client() as client, authenticated_user(client):
            resp = client.get(
                "/api/queries/search", query_string={"q": "betaraven"}
            )
            self.assertEqual(resp.status_code, 200)
            results = json.loads(resp.data)
            matched_ids = set(it["id"] for it in results)

            self.assertIn(name_only.id, matched_ids)
            self.assertIn(desc_only.id, matched_ids)
            self.assertNotIn(sql_only.id, matched_ids)
            self.assertNotIn(neither.id, matched_ids)

    def test_model_search_contract_and_operations(self):
        self.assertTrue(callable(getattr(models.Query, "search", None)))

        q_name = query_factory.create(
            name="GammaLynx name token", description="General info"
        )
        q_desc = query_factory.create(
            name="General title", description="GammaLynx desc token"
        )
        q_null = query_factory.create(
            name="GammaLynx null description", description=None
        )
        q_both = query_factory.create(
            name="GammaLynx both token", description="GammaLynx in desc"
        )
        q_other = query_factory.create(
            name="DeltaOtter only", description="DeltaOtter only"
        )

        # Basic search: case-insensitivity, substring, OR logic, and null description
        results = list(models.Query.search("gammalynx"))
        res_ids = [q.id for q in results]
        self.assertIn(q_name.id, res_ids)
        self.assertIn(q_desc.id, res_ids)
        self.assertIn(q_null.id, res_ids)
        self.assertIn(q_both.id, res_ids)
        self.assertNotIn(q_other.id, res_ids)

        # No duplicate rows if matching both name and description
        self.assertEqual(res_ids.count(q_both.id), 1)

        # Case-insensitivity: uppercase term matching lowercase fields
        upper_results = list(models.Query.search("GAMMALYNX"))
        upper_ids = [q.id for q in upper_results]
        self.assertIn(q_name.id, upper_ids)
        self.assertIn(q_desc.id, upper_ids)

        # Empty search string matches existing queries
        empty_results = list(models.Query.search(""))
        self.assertIn(q_name, empty_results)

        # Non-matching search string returns empty iterable
        none_results = list(models.Query.search("UnmatchedSearchToken_404"))
        self.assertEqual(len(none_results), 0)


if __name__ == "__main__":
    unittest.main()
