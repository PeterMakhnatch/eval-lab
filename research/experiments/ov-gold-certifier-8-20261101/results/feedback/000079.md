class TestCustomCircuitContract(unittest.TestCase):
    """Verifies CustomCircuit initialization, validation, naming, and prediction."""

    def test_invalid_names_rejected_at_initialization_or_predict(self):
        frequencies = np.array([1.0, 10.0])
        invalid_circuits = [
            "Aa0-R1",
            "R2D0-R1",
            "Ra0-R1",
            "RR2D0-R1",
            "rr0-R1",
            "R0-c1",
            "RRA0-R1",
            "RZ0-R1",
            "RRX12-R1",
            "RA0-R1",
        ]
        for c_str in invalid_circuits:
            try:
                model = CustomCircuit(c_str, initial_guess=[1.0, 1.0])
                with self.assertRaises(Exception):
                    model.predict(frequencies, use_initial=True)
            except Exception:
                pass  # Raising during __init__ is valid

    def test_initial_guess_length_mismatch_rejected(self):
        # RR0 needs exactly 1 parameter
        with self.assertRaises(AssertionError):
            CustomCircuit("RR0", initial_guess=[1.0, 2.0])
        with self.assertRaises(AssertionError):
            CustomCircuit("RR0", initial_guess=[])

    def test_public_param_names_and_units_preservation(self):
        model = CustomCircuit(
            "R0-p(RR0,C1)-p(R1,C2032478)-W1",
            initial_guess=[1.0, 0.01, 2.0, 0.02, 3.0, 4.0, 5.0],
        )
        names, units = model.get_param_names()
        self.assertEqual(
            names, ["R0", "RR0", "C1", "R1", "C2032478", "W1_0", "W1_1"]
        )
        self.assertEqual(
            units, ["Ohm", "Ohm", "F", "Ohm", "F", "Ohm", "sec"]
        )

    def test_public_param_names_with_constants_excluded(self):
        model = CustomCircuit(
            "RR0-R1", initial_guess=[0.1], constants={"R1": 10.0}
        )
        names, units = model.get_param_names()
        self.assertEqual(names, ["RR0"])
        self.assertEqual(units, ["Ohm"])

    def test_rr_predict_matches_scaled_resistor_formula(self):
        frequencies = np.array([0.1, 1.0, 10.0, 100.0, 1000.0])
        for val in [0.05, 0.23, 1.0, 3.7]:
            rr_model = CustomCircuit("RR0", initial_guess=[val])
            r_model = CustomCircuit("R0", initial_guess=[100.0 * val])
            z_rr = rr_model.predict(frequencies, use_initial=True)
            z_r = r_model.predict(frequencies, use_initial=True)
            np.testing.assert_array_equal(z_rr, z_r)
            np.testing.assert_array_equal(
                z_rr, np.full_like(frequencies, 100.0 * val + 0j)
            )

    def test_rr_predict_in_parallel_circuit(self):
        frequencies = np.logspace(0, 4, 16)
        omega = 2.0 * np.pi * frequencies
        rr_val = 0.5
        c_val = 1e-4

        model = CustomCircuit("p(RR0,C1)", initial_guess=[rr_val, c_val])
        predicted = model.predict(frequencies, use_initial=True)

        # Independent formula: Z_rr = 100 * rr_val, Z_c = 1 / (j * omega * C)
        z_rr = 100.0 * rr_val
        z_c = 1.0 / (1j * omega * c_val)
        expected = (z_rr * z_c) / (z_rr + z_c)
        np.testing.assert_allclose(predicted, expected, rtol=1e-7, atol=1e-7)

    def test_predict_use_initial_flag_behavior(self):
        frequencies = np.array([1.0, 10.0])
        model = CustomCircuit("RR0", initial_guess=[1.0])
        model.parameters_ = np.array([2.5])
        # With use_initial=True, should use 1.0 -> 100.0
        np.testing.assert_array_equal(
            model.predict(frequencies, use_initial=True),
            np.full_like(frequencies, 100.0 + 0j),
        )
        # With use_initial=False and model fit, should use 2.5 -> 250.0
        np.testing.assert_array_equal(
            model.predict(frequencies, use_initial=False),
            np.full_like(frequencies, 250.0 + 0j),
        )

    def test_predict_rejects_non_ndarray_input(self):
        model = CustomCircuit("RR0", initial_guess=[1.0])
        with self.assertRaises(AssertionError):
            model.predict([1.0, 10.0])


class TestFittingAndBoundsContract(unittest.TestCase):
    """Verifies default bounds generation, constants exclusion, and SciPy optimization."""

    def test_constants_skipped_in_default_bounds_fitting(self):
        # Fitting a circuit with fixed parameters must skip bounds for constants.
        frequencies = np.logspace(-1, 4, 32)
        r_const = 15.0
        rr_true = 0.35
        target_impedance = np.full(
            frequencies.shape, 100.0 * rr_true + r_const + 0j
        )

        model = CustomCircuit(
            circuit="RR0-R1", initial_guess=[0.1], constants={"R1": r_const}
        )
        # Default bounds (bounds=None) must not raise length mismatch error
        model.fit(frequencies, target_impedance)
        np.testing.assert_allclose(
            model.parameters_, [rr_true], rtol=1e-4, atol=1e-7
        )

    def test_warburg_upper_bound_allows_parameter_one_above_one(self):
        # For non-E elements (e.g. Warburg W), parameter index 1 must have ub = inf, not 1.
        frequencies = np.logspace(0, 3, 24)
        true_w_params = [50.0, 12.0]  # Second parameter is 12.0 (> 1.0)
        true_model = CustomCircuit("W0", initial_guess=true_w_params)
        target_impedance = true_model.predict(frequencies, use_initial=True)

        fit_model = CustomCircuit("W0", initial_guess=[40.0, 8.0])
        fit_model.fit(frequencies, target_impedance)

        np.testing.assert_allclose(
            fit_model.parameters_, true_w_params, rtol=1e-3, atol=1e-5
        )
        self.assertGreater(fit_model.parameters_[1], 1.0)

    def test_cpe_alpha_upper_bound_is_one(self):
        # CPE element E at index 1 (alpha) must have upper bound of 1.0
        frequencies = np.logspace(0, 4, 24)
        true_e_params = [1e-4, 0.85]
        true_model = CustomCircuit("E0", initial_guess=true_e_params)
        target_impedance = true_model.predict(frequencies, use_initial=True)

        fit_model = CustomCircuit("E0", initial_guess=[5e-5, 0.6])
        fit_model.fit(frequencies, target_impedance)
        np.testing.assert_allclose(
            fit_model.parameters_, true_e_params, rtol=1e-3, atol=1e-5
        )
        self.assertLessEqual(fit_model.parameters_[1], 1.0)

    def test_multiletter_multidigit_fit_convergence(self):
        frequencies = np.logspace(-1, 4, 30)
        rr_true = 0.45
        c_true = 1.2e-4
        target_z = 100.0 * rr_true + 1.0 / (
            2j * np.pi * frequencies * c_true
        )

        model = CustomCircuit(
            "RR99-C8888", initial_guess=[0.2, 5e-5]
        )
        model.fit(frequencies, target_z)
        np.testing.assert_allclose(
            model.parameters_, [rr_true, c_true], rtol=5e-3, atol=1e-7
        )

    def test_direct_circuit_fit_wrapper(self):
        frequencies = np.logspace(0, 2, 10)
        target_z = np.full(frequencies.shape, 300.0 + 0j)
        popt, pcov = circuit_fit(
            frequencies, target_z, "RR0", [1.0]
        )
        np.testing.assert_allclose(popt, [3.0], rtol=1e-4)


class TestModelPersistenceAndString(unittest.TestCase):
    """Verifies serialization, pretty printing, and regression on standard models."""

    def test_custom_circuit_str_representation(self):
        model = CustomCircuit(
            "RR0-R1", initial_guess=[0.25], constants={"R1": 10.0}
        )
        s = str(model)
        self.assertIn("RR0", s)
        self.assertIn("R1", s)
        self.assertIn("Circuit string: RR0-R1", s)
        self.assertIn("Ohm", s)

    def test_json_save_and_load_roundtrip(self):
        frequencies = np.array([1.0, 10.0, 100.0])
        model = CustomCircuit("RR10-C20", initial_guess=[0.5, 1e-4])
        model.parameters_ = np.array([0.5, 1e-4])
        model.conf_ = np.array([0.01, 1e-6])

        with tempfile.NamedTemporaryFile(suffix=".json", delete=False) as tmp:
            tmp_path = tmp.name

        try:
            model.save(tmp_path)
            loaded = CustomCircuit()
            loaded.load(tmp_path)
            self.assertEqual(loaded.circuit, "RR10-C20")
            np.testing.assert_allclose(loaded.parameters_, [0.5, 1e-4])
            np.testing.assert_allclose(loaded.conf_, [0.01, 1e-6])
            np.testing.assert_allclose(
                loaded.predict(frequencies), model.predict(frequencies)
            )

            loaded_initial = CustomCircuit()
            loaded_initial.load(tmp_path, fitted_as_initial=True)
            np.testing.assert_allclose(loaded_initial.initial_guess, [0.5, 1e-4])
        finally:
            if os.path.exists(tmp_path):
                os.remove(tmp_path)

    def test_randles_circuits_regression(self):
        r_model = Randles(initial_guess=[10.0, 50.0, 1e-5, 20.0])
        self.assertEqual(r_model.circuit, "R0-p(R1,C1)-W1")
        self.assertEqual(len(r_model.get_param_names()[0]), 5)

        r_cpe = Randles(CPE=True, initial_guess=[10.0, 50.0, 1e-5, 0.9])
        self.assertEqual(r_cpe.circuit, "R0-p(R1,E1)-W1")
        self.assertEqual(len(r_cpe.get_param_names()[0]), 5)
