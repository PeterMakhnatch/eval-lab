import sys
sys.path.insert(0, '/testbed')

import unittest
import numpy as np
from scipy.stats import norm

from robo.acquisition.LogEI import LogEI

SENTINEL = -777.0


class StubModel(object):
    """Deterministic model: mean(x) = x, constant predictive variance.

    A query at the sentinel point (the incumbent location) returns the
    incumbent's objective value as a plain (n, 1) array.
    """

    def __init__(self, var, eta):
        self.var = var
        self.eta = eta

    def predict(self, X, **kwargs):
        X = np.asarray(X, dtype=float)
        if X.size and X.ravel()[0] == SENTINEL:
            return np.ones((X.shape[0], 1)) * self.eta
        m = X[:, :1].copy()
        v = np.ones(m.shape) * self.var
        return m, v


def evaluate(var, eta, mu):
    model = StubModel(var, eta)
    X_lower = np.array([-10.0])
    X_upper = np.array([10.0])

    def inc(m):
        return np.array([SENTINEL])

    acq = LogEI(model, X_lower, X_upper, compute_incumbent=inc, par=0.0)
    old = np.seterr(all='ignore')
    try:
        res = acq(np.array([[mu]]))
    finally:
        np.seterr(**old)
    return float(np.ravel(res)[0])


def true_log_ei(mu, sigma, eta):
    z = (eta - mu) / sigma
    ei = (eta - mu) * norm.cdf(z) + sigma * norm.pdf(z)
    return np.log(ei)


class LogEIBehaviorTest(unittest.TestCase):

    def test_matches_closed_form_log_ei(self):
        # (mu, eta, var): mixes mean below and above the incumbent value
        cases = [
            (0.0, 1.5, 1.0),
            (0.5, 2.0, 4.0),
            (1.0, 0.5, 4.0),
            (1.0, 0.5, 1.0),
            (2.0, 0.5, 1.0),
            (3.0, 0.5, 4.0),
        ]
        for mu, eta, var in cases:
            val = evaluate(var, eta, mu)
            expected = true_log_ei(mu, np.sqrt(var), eta)
            self.assertTrue(np.isfinite(val),
                            "non-finite for mu=%s eta=%s var=%s: %r" % (mu, eta, var, val))
            self.assertAlmostEqual(val, expected, places=6)

    def test_mean_exactly_at_unit_boundary(self):
        # mean 1.0, incumbent value 0.5, sigma 2: improvement is negative in mean
        # but the log EI must still be the exact log of the closed form
        val = evaluate(4.0, 0.5, 1.0)
        self.assertTrue(np.isfinite(val))
        self.assertAlmostEqual(val, true_log_ei(1.0, 2.0, 0.5), places=6)

    def test_zero_variance_mean_not_better_gives_neg_inf(self):
        for mu in (1.0, 2.0):
            val = evaluate(0.0, 0.5, mu)
            self.assertTrue(np.isinf(val) and val < 0,
                            "expected -inf for mu=%s, got %r" % (mu, val))

    def test_zero_variance_mean_better_gives_log_improvement(self):
        val = evaluate(0.0, 1.5, 0.0)
        self.assertAlmostEqual(val, np.log(1.5), places=6)

    def test_mean_equal_incumbent_gives_log_sigma_pdf(self):
        # when the mean equals the incumbent value, EI = sigma * pdf(0)
        for var, level in ((1.0, 1.0), (4.0, 4.0), (4.0, 1.0)):
            sigma = np.sqrt(var)
            val = evaluate(var, level, level)
            expected = np.log(sigma) + norm.logpdf(0.0)
            self.assertTrue(np.isfinite(val))
            self.assertAlmostEqual(val, expected, places=6)


if __name__ == '__main__':
    unittest.main()
