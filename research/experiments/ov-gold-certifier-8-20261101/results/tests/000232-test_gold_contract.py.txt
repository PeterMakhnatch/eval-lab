import sys
sys.path.insert(0, '/testbed')

import unittest
import numpy as np
import quantecon as qe
from quantecon.gridtools import cartesian, cartesian_nearest_index


class TestCartesianNearestIndexInterface(unittest.TestCase):
    def test_public_exports(self):
        self.assertTrue(callable(qe.cartesian_nearest_index))
        self.assertTrue(callable(cartesian_nearest_index))
        self.assertIs(qe.cartesian_nearest_index, cartesian_nearest_index)

    def test_docstring_examples(self):
        nodes = (np.arange(3), np.arange(2))
        x_scalar = (0.6, 0.4)
        self.assertEqual(qe.cartesian_nearest_index(x_scalar, nodes), 2)

        x_batch = [(-0.1, 1.2), (2, 0)]
        actual = qe.cartesian_nearest_index(x_batch, nodes)
        np.testing.assert_array_equal(actual, np.array([1, 4]))

    def test_issue_description_example(self):
        nodes = (np.arange(3), np.arange(2))
        idx = qe.cartesian_nearest_index((0.6, 0.4), nodes)
        self.assertEqual(idx, 2)

    def test_return_type_1d_query(self):
        nodes = (np.linspace(0, 1, 5), np.linspace(0, 1, 6))
        for x in [(0.2, 0.8), [0.2, 0.8], np.array([0.2, 0.8])]:
            res = qe.cartesian_nearest_index(x, nodes)
            self.assertTrue(np.isscalar(res) or isinstance(res, (int, np.integer)))
            self.assertFalse(isinstance(res, np.ndarray))

    def test_return_type_2d_query(self):
        nodes = (np.linspace(0, 1, 5), np.linspace(0, 1, 6))
        for X in [
            [(-0.1, 0.2), (0.5, 0.5), (1.1, 0.9)],
            np.array([[-0.1, 0.2], [0.5, 0.5], [1.1, 0.9]]),
        ]:
            res = qe.cartesian_nearest_index(X, nodes)
            self.assertIsInstance(res, np.ndarray)
            self.assertEqual(res.ndim, 1)
            self.assertEqual(len(res), 3)
            self.assertTrue(np.issubdtype(res.dtype, np.integer))

    def test_batch_size_one_returns_1d_array(self):
        nodes = (np.arange(3), np.arange(2))
        X = np.array([[0.6, 0.4]])
        res = qe.cartesian_nearest_index(X, nodes)
        self.assertIsInstance(res, np.ndarray)
        self.assertEqual(res.shape, (1,))
        self.assertEqual(res[0], 2)

    def test_batch_size_zero_returns_empty_1d_array(self):
        nodes = (np.arange(3), np.arange(2))
        X = np.empty((0, 2))
        res = qe.cartesian_nearest_index(X, nodes)
        self.assertIsInstance(res, np.ndarray)
        self.assertEqual(res.shape, (0,))
        self.assertTrue(np.issubdtype(res.dtype, np.integer))

    def test_input_containers_for_nodes(self):
        x = (1.2, 0.3)
        expected = 2

        # tuple of numpy arrays
        nodes_tuple_arr = (np.array([0, 1, 2]), np.array([0, 1]))
        self.assertEqual(qe.cartesian_nearest_index(x, nodes_tuple_arr), expected)

        # list of numpy arrays
        nodes_list_arr = [np.array([0, 1, 2]), np.array([0, 1])]
        self.assertEqual(qe.cartesian_nearest_index(x, nodes_list_arr), expected)

        # tuple of lists
        nodes_tuple_list = ([0, 1, 2], [0, 1])
        self.assertEqual(qe.cartesian_nearest_index(x, nodes_tuple_list), expected)

        # list of lists
        nodes_list_list = [[0, 1, 2], [0, 1]]
        self.assertEqual(qe.cartesian_nearest_index(x, nodes_list_list), expected)

        # tuple of range objects
        nodes_ranges = (range(3), range(2))
        self.assertEqual(qe.cartesian_nearest_index(x, nodes_ranges), expected)


class TestCartesianNearestIndexValidation(unittest.TestCase):
    def setUp(self):
        self.nodes = (np.linspace(0, 1, 5), np.linspace(0, 1, 6))

    def test_dimension_mismatch_1d_too_short(self):
        with self.assertRaises(ValueError) as ctx:
            qe.cartesian_nearest_index((0.5,), self.nodes)
        self.assertIn("point `x` must have same length as `nodes`", str(ctx.exception))

    def test_dimension_mismatch_1d_too_long(self):
        with self.assertRaises(ValueError) as ctx:
            qe.cartesian_nearest_index((0.5, 0.5, 0.5), self.nodes)
        self.assertIn("point `x` must have same length as `nodes`", str(ctx.exception))

    def test_dimension_mismatch_2d_too_short(self):
        X = np.ones((4, 1))
        with self.assertRaises(ValueError) as ctx:
            qe.cartesian_nearest_index(X, self.nodes)
        self.assertIn("points in `x` must have same length as `nodes`", str(ctx.exception))

    def test_dimension_mismatch_2d_too_long(self):
        X = np.ones((4, 3))
        with self.assertRaises(ValueError) as ctx:
            qe.cartesian_nearest_index(X, self.nodes)
        self.assertIn("points in `x` must have same length as `nodes`", str(ctx.exception))

    def test_passing_materialized_product_as_nodes_raises(self):
        prod = cartesian(self.nodes)
        with self.assertRaises(ValueError):
            qe.cartesian_nearest_index((0.5, 0.5), prod)

        with self.assertRaises(ValueError):
            qe.cartesian_nearest_index(np.array([[0.5, 0.5]]), prod)


class TestCartesianNearestIndexBoundariesAndTies(unittest.TestCase):
    def test_exact_lower_bound(self):
        nodes_1d = ([1.0, 2.0, 3.0],)
        self.assertEqual(qe.cartesian_nearest_index((1.0,), nodes_1d), 0)

        nodes_2d = ([1.0, 2.0, 3.0], [10.0, 20.0])
        self.assertEqual(qe.cartesian_nearest_index((1.0, 10.0), nodes_2d), 0)
        self.assertEqual(qe.cartesian_nearest_index((1.0, 20.0), nodes_2d), 1)

        nodes_3d = ([1.0, 2.0], [10.0, 20.0], [100.0, 200.0])
        self.assertEqual(qe.cartesian_nearest_index((1.0, 10.0, 100.0), nodes_3d), 0)

    def test_strictly_below_lower_bound(self):
        nodes = ([1.0, 2.0, 3.0], [10.0, 20.0])
        self.assertEqual(qe.cartesian_nearest_index((-100.0, -50.0), nodes), 0)
        self.assertEqual(qe.cartesian_nearest_index((0.999, 10.0), nodes), 0)

    def test_exact_upper_bound(self):
        nodes_1d = ([1.0, 2.0, 3.0],)
        self.assertEqual(qe.cartesian_nearest_index((3.0,), nodes_1d), 2)

        nodes_2d = ([1.0, 2.0, 3.0], [10.0, 20.0])
        self.assertEqual(qe.cartesian_nearest_index((3.0, 20.0), nodes_2d, order='C'), 5)
        self.assertEqual(qe.cartesian_nearest_index((3.0, 20.0), nodes_2d, order='F'), 5)

        nodes_3d = ([1.0, 2.0], [10.0, 20.0], [100.0, 200.0])
        self.assertEqual(qe.cartesian_nearest_index((2.0, 20.0, 200.0), nodes_3d), 7)

    def test_strictly_above_upper_bound(self):
        nodes = ([1.0, 2.0, 3.0], [10.0, 20.0])
        self.assertEqual(qe.cartesian_nearest_index((100.0, 200.0), nodes, order='C'), 5)
        self.assertEqual(qe.cartesian_nearest_index((3.001, 20.001), nodes, order='C'), 5)

    def test_tie_breaking_midpoint_contracts(self):
        # 1D single interval tie: 5.0 is equidistant to 0.0 and 10.0
        # Contract: ties break toward lower index (k-1), matching np.argmin
        nodes_1d = ([0.0, 10.0],)
        self.assertEqual(qe.cartesian_nearest_index((5.0,), nodes_1d), 0)

        # 1D multiple intervals tie
        nodes_multi = ([0.0, 10.0, 20.0, 30.0],)
        self.assertEqual(qe.cartesian_nearest_index((5.0,), nodes_multi), 0)
        self.assertEqual(qe.cartesian_nearest_index((15.0,), nodes_multi), 1)
        self.assertEqual(qe.cartesian_nearest_index((25.0,), nodes_multi), 2)

        # 2D ties
        nodes_2d = ([0.0, 2.0], [0.0, 2.0])
        self.assertEqual(qe.cartesian_nearest_index((1.0, 1.0), nodes_2d, order='C'), 0)
        self.assertEqual(qe.cartesian_nearest_index((1.0, 1.0), nodes_2d, order='F'), 0)
        self.assertEqual(qe.cartesian_nearest_index((1.0, 0.0), nodes_2d, order='C'), 0)
        self.assertEqual(qe.cartesian_nearest_index((0.0, 1.0), nodes_2d, order='C'), 0)

    def test_near_midpoint_perturbation(self):
        nodes = ([0.0, 10.0],)
        self.assertEqual(qe.cartesian_nearest_index((4.999,), nodes), 0)
        self.assertEqual(qe.cartesian_nearest_index((5.001,), nodes), 1)

    def test_exact_interior_grid_points(self):
        nodes = ([0.0, 1.0, 2.0, 3.0], [10.0, 20.0, 30.0])
        self.assertEqual(qe.cartesian_nearest_index((2.0, 20.0), nodes, order='C'), 7)
        self.assertEqual(qe.cartesian_nearest_index((1.0, 30.0), nodes, order='C'), 5)


class TestCartesianNearestIndexSingletonAndDtypes(unittest.TestCase):
    def test_singleton_dimensions(self):
        # Singleton in first dimension
        nodes_first_singleton = ([2.5], np.array([-1.0, 1.0]))
        self.assertEqual(qe.cartesian_nearest_index((2.5, 0.0), nodes_first_singleton), 0)
        self.assertEqual(qe.cartesian_nearest_index((-100.0, 9.0), nodes_first_singleton), 1)
        actual_first = qe.cartesian_nearest_index(np.array([[2.5, -4.0], [2.5, 4.0]]), nodes_first_singleton)
        np.testing.assert_array_equal(actual_first, np.array([0, 1]))

        # Singleton in second dimension
        nodes_second_singleton = (np.array([-1.0, 1.0]), [2.5])
        self.assertEqual(qe.cartesian_nearest_index((0.0, 2.5), nodes_second_singleton), 0)
        self.assertEqual(qe.cartesian_nearest_index((9.0, -100.0), nodes_second_singleton), 1)
        actual_second = qe.cartesian_nearest_index(np.array([[-4.0, 2.5], [4.0, 2.5]]), nodes_second_singleton)
        np.testing.assert_array_equal(actual_second, np.array([0, 1]))

        # Singleton in all dimensions
        nodes_all_singleton = ([3.0], [4.0])
        self.assertEqual(qe.cartesian_nearest_index((-10.0, -10.0), nodes_all_singleton), 0)
        self.assertEqual(qe.cartesian_nearest_index((10.0, 10.0), nodes_all_singleton), 0)

        # 1-dimensional singleton
        nodes_1d_singleton = ([42.0],)
        self.assertEqual(qe.cartesian_nearest_index((42.0,), nodes_1d_singleton), 0)
        self.assertEqual(qe.cartesian_nearest_index((-10.0,), nodes_1d_singleton), 0)
        self.assertEqual(qe.cartesian_nearest_index((100.0,), nodes_1d_singleton), 0)

    def test_mixed_dtypes_promotion(self):
        nodes = (
            np.array([0, 1, 2], dtype=np.int32),
            np.array([0.25, 0.75], dtype=np.float64),
        )
        self.assertEqual(qe.cartesian_nearest_index((0, 0.75), nodes), 1)
        self.assertEqual(qe.cartesian_nearest_index((1, 0.25), nodes), 2)

    def test_python_mixed_types_promotion(self):
        nodes = ([0, 1, 2], [0.2, 0.8])
        self.assertEqual(qe.cartesian_nearest_index((0.0, 0.8), nodes), 1)
        self.assertEqual(qe.cartesian_nearest_index((2.0, 0.2), nodes), 4)

    def test_integer_query_coordinates(self):
        nodes = ([0.0, 1.0, 2.0], [0.0, 1.0, 2.0])
        self.assertEqual(qe.cartesian_nearest_index((1, 2), nodes), 5)


class TestCartesianNearestIndexOrderAndDimensions(unittest.TestCase):
    def test_order_default_matches_C(self):
        nodes = ([0.0, 1.0], [10.0, 20.0, 30.0])
        x = (1.0, 10.0)
        self.assertEqual(qe.cartesian_nearest_index(x, nodes), 3)
        self.assertEqual(qe.cartesian_nearest_index(x, nodes, order='C'), 3)

        X = np.array([[0.0, 10.0], [1.0, 10.0], [0.0, 30.0]])
        np.testing.assert_array_equal(
            qe.cartesian_nearest_index(X, nodes),
            qe.cartesian_nearest_index(X, nodes, order='C'),
        )

    def test_order_C_vs_F_asymmetric_grid(self):
        nodes = ([0.0, 1.0], [10.0, 20.0, 30.0])
        # prod in 'C': (0,10)->0, (0,20)->1, (0,30)->2, (1,10)->3, (1,20)->4, (1,30)->5
        # prod in 'F': (0,10)->0, (1,10)->1, (0,20)->2, (1,20)->3, (0,30)->4, (1,30)->5
        points_and_expected = [
            ((0.0, 10.0), 0, 0),
            ((1.0, 10.0), 3, 1),
            ((0.0, 20.0), 1, 2),
            ((1.0, 20.0), 4, 3),
            ((0.0, 30.0), 2, 4),
            ((1.0, 30.0), 5, 5),
        ]
        for pt, exp_c, exp_f in points_and_expected:
            self.assertEqual(qe.cartesian_nearest_index(pt, nodes, order='C'), exp_c)
            self.assertEqual(qe.cartesian_nearest_index(pt, nodes, order='F'), exp_f)

        queries = np.array([pt for pt, _, _ in points_and_expected])
        expected_c = np.array([exp_c for _, exp_c, _ in points_and_expected])
        expected_f = np.array([exp_f for _, _, exp_f in points_and_expected])
        np.testing.assert_array_equal(qe.cartesian_nearest_index(queries, nodes, order='C'), expected_c)
        np.testing.assert_array_equal(qe.cartesian_nearest_index(queries, nodes, order='F'), expected_f)

    def test_order_keyword_and_positional(self):
        nodes = ([0.0, 1.0], [10.0, 20.0, 30.0])
        x = (1.0, 10.0)
        self.assertEqual(
            qe.cartesian_nearest_index(x, nodes, 'F'),
            qe.cartesian_nearest_index(x, nodes, order='F'),
        )

    def test_all_product_points_recovery_1d(self):
        nodes = (np.linspace(-2.0, 2.0, 11),)
        for order in ('C', 'F'):
            prod = cartesian(nodes, order=order)
            actual_batch = qe.cartesian_nearest_index(prod, nodes, order=order)
            np.testing.assert_array_equal(actual_batch, np.arange(len(prod)))
            for k in range(len(prod)):
                self.assertEqual(qe.cartesian_nearest_index(prod[k], nodes, order=order), k)

    def test_all_product_points_recovery_2d(self):
        nodes = (np.arange(3), np.arange(4))
        for order in ('C', 'F'):
            prod = cartesian(nodes, order=order)
            actual_batch = qe.cartesian_nearest_index(prod, nodes, order=order)
            np.testing.assert_array_equal(actual_batch, np.arange(len(prod)))
            for k in range(len(prod)):
                self.assertEqual(qe.cartesian_nearest_index(prod[k], nodes, order=order), k)

    def test_all_product_points_recovery_3d(self):
        nodes = (np.linspace(0, 1, 3), np.linspace(0, 2, 4), np.linspace(-1, 1, 2))
        for order in ('C', 'F'):
            prod = cartesian(nodes, order=order)
            actual_batch = qe.cartesian_nearest_index(prod, nodes, order=order)
            np.testing.assert_array_equal(actual_batch, np.arange(len(prod)))
            for k in range(len(prod)):
                self.assertEqual(qe.cartesian_nearest_index(prod[k], nodes, order=order), k)

    def test_all_product_points_recovery_4d(self):
        nodes = (np.arange(2), np.arange(3), np.arange(2), np.arange(3))
        for order in ('C', 'F'):
            prod = cartesian(nodes, order=order)
            actual_batch = qe.cartesian_nearest_index(prod, nodes, order=order)
            np.testing.assert_array_equal(actual_batch, np.arange(len(prod)))
            for k in range(len(prod)):
                self.assertEqual(qe.cartesian_nearest_index(prod[k], nodes, order=order), k)


class TestCartesianNearestIndexAgainstLinearSearch(unittest.TestCase):
    def linear_search(self, nodes, x, order='C'):
        x = np.asarray(x)
        prod = cartesian(nodes, order=order)
        return int(((prod - x)**2).sum(axis=1).argmin())

    def linear_search_batch(self, nodes, X, order='C'):
        prod = cartesian(nodes, order=order)
        out = np.empty(len(X), dtype=np.intp)
        for t in range(len(X)):
            out[t] = ((prod - X[t])**2).sum(axis=1).argmin()
        return out

    def test_random_queries_1d_grid(self):
        rng = np.random.default_rng(1001)
        nodes = (np.linspace(-3.0, 3.0, 25),)
        T = 30
        X = rng.uniform(-5.0, 5.0, size=(T, 1))

        for order in ('C', 'F'):
            expected = self.linear_search_batch(nodes, X, order=order)
            actual = qe.cartesian_nearest_index(X, nodes, order=order)
            np.testing.assert_array_equal(actual, expected)

            for t in range(T):
                self.assertEqual(
                    qe.cartesian_nearest_index(X[t], nodes, order=order),
                    expected[t],
                )

    def test_random_queries_2d_grid(self):
        rng = np.random.default_rng(1002)
        nodes = (np.linspace(-1.0, 2.0, 7), np.linspace(10.0, 20.0, 8))
        T = 40
        X = np.column_stack((
            rng.uniform(-3.0, 4.0, size=T),
            rng.uniform(8.0, 22.0, size=T),
        ))

        for order in ('C', 'F'):
            expected = self.linear_search_batch(nodes, X, order=order)
            actual = qe.cartesian_nearest_index(X, nodes, order=order)
            np.testing.assert_array_equal(actual, expected)

            for t in range(T):
                self.assertEqual(
                    qe.cartesian_nearest_index(X[t], nodes, order=order),
                    expected[t],
                )

    def test_random_queries_3d_grid(self):
        rng = np.random.default_rng(1003)
        nodes = (np.linspace(0, 1, 5), np.linspace(-2, 2, 4), np.linspace(10, 15, 3))
        T = 30
        X = np.column_stack((
            rng.uniform(-0.5, 1.5, size=T),
            rng.uniform(-3.0, 3.0, size=T),
            rng.uniform(9.0, 16.0, size=T),
        ))

        for order in ('C', 'F'):
            expected = self.linear_search_batch(nodes, X, order=order)
            actual = qe.cartesian_nearest_index(X, nodes, order=order)
            np.testing.assert_array_equal(actual, expected)

            for t in range(T):
                self.assertEqual(
                    qe.cartesian_nearest_index(X[t], nodes, order=order),
                    expected[t],
                )

    def test_random_queries_4d_grid(self):
        rng = np.random.default_rng(1004)
        nodes = (np.arange(3), np.linspace(0, 1, 4), np.arange(2), np.linspace(-1, 1, 3))
        T = 20
        X = np.column_stack((
            rng.uniform(-1, 4, size=T),
            rng.uniform(-0.5, 1.5, size=T),
            rng.uniform(-1, 3, size=T),
            rng.uniform(-2, 2, size=T),
        ))

        for order in ('C', 'F'):
            expected = self.linear_search_batch(nodes, X, order=order)
            actual = qe.cartesian_nearest_index(X, nodes, order=order)
