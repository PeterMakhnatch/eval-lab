import sys
sys.path.insert(0, '/testbed')

import unittest
from io import StringIO

try:
    from unittest import mock
except ImportError:
    import mock

import gpustat
from gpustat.test_gpustat import _configure_mock


def _render_fan_line(N, Process, value):
    _configure_mock(N, Process)
    stats = gpustat.new_query()
    stats[0].entry['fan.speed'] = value
    fp = StringIO()
    stats.print_formatted(fp=fp, force_color=True, show_fan=True,
                          show_header=False)
    line = [l for l in fp.getvalue().split('\n') if l][0]
    marker = '%3d %%' % value
    assert marker in line, (marker, line)
    # Neutralise the numeric text, leaving only the styling escape codes.
    return line.replace(marker, 'NNN %')


class TestFanColorThreshold(unittest.TestCase):

    @mock.patch('psutil.Process')
    @mock.patch('gpustat.core.N')
    def test_fan_exactly_50_uses_high_style(self, N, Process):
        low = _render_fan_line(N, Process, 49)
        mid = _render_fan_line(N, Process, 50)
        high = _render_fan_line(N, Process, 51)
        # Below 50 is styled differently from 50 and above.
        self.assertNotEqual(low, mid)
        self.assertNotEqual(low, high)
        # 50 is in the same (high) category as 51.
        self.assertEqual(mid, high)

    @mock.patch('psutil.Process')
    @mock.patch('gpustat.core.N')
    def test_fan_below_50_same_style(self, N, Process):
        a = _render_fan_line(N, Process, 10)
        b = _render_fan_line(N, Process, 49)
        self.assertEqual(a, b)


if __name__ == '__main__':
    unittest.main()
