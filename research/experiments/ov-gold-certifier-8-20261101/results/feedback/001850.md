def memory_object_count(name: str) -> int:
    return duckdb.execute(
        "SELECT count(*) FROM information_schema.tables "
        "WHERE table_catalog = 'memory' AND table_schema = 'main' "
        "AND table_name = ?",
        [name],
    ).fetchone()[0]

        CREATE TABLE "my_table_with_select" AS SELECT 1;

        CREATE TABLE "with a space" (id INT);

        CREATE TABLE e'escaped\ntable' (id INT);

        ATTACH 'test1.db' AS 'single_schema';
        ATTACH 'test2.db' AS e'escaped\nschema';
        """

        self.assertEqual(

        # Escape string literal contains keywords like FROM, WHERE, CREATE TABLE,

            e'CREATE TABLE inner_fake (id INT)' AS col3,
            e'' AS empty_literal

        self.assertNotIn("inner_fake", defs.tables)

class TestRealDuckDBExecutionAndState(unittest.TestCase):
    def setUp(self):
        self.kernel = Kernel.__new__(Kernel)
        self.kernel._module = ModuleType("mimo_real_duckdb_" + uuid4().hex)

    def test_real_table_invalidation_and_exclusion(self):
        name = "tbl_" + uuid4().hex
        sql = f"CREATE TABLE {name} AS SELECT e'escaped\\tval' AS val"
        cell = compile_sql_cell(sql, cell_id="cell_table")
        self.assertIn(name, cell.variable_data)

        try:
            duckdb.execute(sql)
            self.assertEqual(
                duckdb.execute(f"SELECT val FROM memory.main.{name}").fetchall(),
                [("escaped\tval",)],
            )

            # Excluded table must survive
            self.kernel._delete_variables(cell.variable_data, excluded={name})
            self.assertEqual(memory_object_count(name), 1)

            # Non-excluded table must be dropped
            self.kernel._delete_variables(cell.variable_data, excluded=set())
            self.assertEqual(memory_object_count(name), 0)
        finally:
            duckdb.execute(f"DROP TABLE IF EXISTS memory.main.{name}")

    def test_real_view_invalidation(self):
        name = "vw_" + uuid4().hex
        sql = f"CREATE VIEW {name} AS SELECT e'newline\\nval' AS val"
        cell = compile_sql_cell(sql, cell_id="cell_view")
        self.assertIn(name, cell.variable_data)

        try:
            duckdb.execute(sql)
            self.assertEqual(
                duckdb.execute(f"SELECT val FROM memory.main.{name}").fetchall(),
                [("newline\nval",)],
            )
            self.kernel._delete_variables(cell.variable_data, excluded=set())
            self.assertEqual(memory_object_count(name), 0)
        finally:
            duckdb.execute(f"DROP VIEW IF EXISTS memory.main.{name}")

    def test_real_attached_database_detach(self):
        alias = "db_" + uuid4().hex
        with tempfile.TemporaryDirectory() as temp_dir:
            db_path = str(Path(temp_dir) / "test.duckdb").replace("'", "''")
            sql = f"ATTACH '{db_path}' AS {alias}"
            cell = compile_sql_cell(sql, cell_id="cell_attach")
            self.assertIn(alias, cell.variable_data)

            try:
                duckdb.execute(sql)
                dbs = [row[1] for row in duckdb.execute("PRAGMA database_list").fetchall()]
                self.assertIn(alias, dbs)

                self.kernel._delete_variables(cell.variable_data, excluded=set())
                dbs_after = [row[1] for row in duckdb.execute("PRAGMA database_list").fetchall()]
                self.assertNotIn(alias, dbs_after)
            finally:
                duckdb.execute(f"DETACH DATABASE IF EXISTS {alias}")
