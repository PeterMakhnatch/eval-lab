## marimo mishandles SQL containing DuckDB escape string literals (`e'...'`)

I'm using marimo with DuckDB and ran into a bug when my SQL contains DuckDB's `e'...'` escape string literal — the form that lets you use `\n`, `\t`, etc. inside a string.

Minimal repro. In one cell:

```python
mo.sql(
    """
    CREATE TABLE my_table AS
    SELECT * FROM read_csv(e'/some/path/file\tname.csv')
    """
)
```

In a separate cell:

```python
mo.sql("SELECT * FROM my_table")
```

The SQL itself is perfectly valid DuckDB — if I paste the same statements into a plain DuckDB session they run without any issue.

Inside marimo, though, things don't behave normally for this cell. The relationship between the cell that creates the table and the cell that reads from it doesn't work the way it does for "normal" SQL cells — it feels like marimo isn't seeing this cell the same way it sees other SQL cells, and the surrounding behavior (reactivity, what happens when I edit/re-run the cell) gets weird.

As a sanity check: if I drop the `e` prefix and just write a plain `'...'` literal (so no escape sequences), everything goes back to working correctly. So the problem seems specific to the `e'...'` form.

It would be great if marimo handled SQL that contains `e'...'` literals the same way it handles any other DuckDB SQL.

#!/usr/bin/env python3
"""Observe real DuckDB state after the submitted Kernel SQL cleanup method."""

from pathlib import Path
import sys
import tempfile
from types import ModuleType
import unittest
from uuid import uuid4

sys.path.insert(0, "/testbed")

import duckdb
from marimo._ast.compiler import compile_cell
from marimo._runtime.runtime import Kernel


def compile_sql_cell(sql, cell_id):
    # Use the actual compiler to obtain the SQL variables a real cell owns.
    # This compiles a complete cell but does not start a notebook/frontend.
    code = "import marimo as mo\nmo.sql(" + repr(sql) + ")"
    return compile_cell(code, cell_id=cell_id, carried_imports=[])


def memory_object_count(name):
    return duckdb.execute(
        "SELECT count(*) FROM information_schema.tables "
        "WHERE table_catalog = 'memory' AND table_schema = 'main' "
        "AND table_name = ?",
        [name],
    ).fetchone()[0]


class SQLCellCleanupBehavior(unittest.TestCase):
    def setUp(self):
        # The cleanup operation needs only Kernel's real module namespace.
        # Keep the actual Kernel object and all its methods unchanged, while
        # avoiding unrelated frontend, stream, watcher and worker startup.
        self.kernel = Kernel.__new__(Kernel)
        self.kernel._module = ModuleType("mimo_cleanup_namespace_" + uuid4().hex)

    def test_table_invalidation_removes_old_data_and_allows_rerun(self):
        for literal, decoded in (
            ("e'left\\tright'", "left\tright"),
            ("e'new\\nline'", "new\nline"),
        ):
            with self.subTest(literal=literal):
                name = "mimo_table_" + uuid4().hex
                sql = "CREATE TABLE " + name + " AS SELECT " + literal + " AS value"
                cell = compile_sql_cell(sql, "table_cell")
                self.assertIn(name, cell.variable_data,
                              "SQL definitions must participate in cell ownership")
                try:
                    duckdb.execute(sql)
                    self.assertEqual(
                        duckdb.execute("SELECT value FROM memory.main." + name).fetchall(),
                        [(decoded,)],
                    )

                    self.kernel._delete_variables(cell.variable_data, {name})
                    self.assertEqual(memory_object_count(name), 1,
                                     "excluded SQL definitions must survive")
                    self.kernel._delete_variables(cell.variable_data, set())
                    self.assertEqual(memory_object_count(name), 0,
                                     "invalidated SQL table still exists")

                    # A rerun must be able to CREATE the same table again and
                    # expose only the replacement cell's rows.
                    duckdb.execute("CREATE TABLE " + name + " AS SELECT 23 AS value")
                    self.assertEqual(
                        duckdb.execute("SELECT value FROM memory.main." + name).fetchall(),
                        [(23,)],
                    )
                    self.kernel._delete_variables(cell.variable_data, set())
                    self.assertEqual(memory_object_count(name), 0)
                finally:
                    duckdb.execute("DROP TABLE IF EXISTS memory.main." + name)

    def test_cleanup_keeps_unrelated_and_persistent_tables(self):
        name = "mimo_owned_" + uuid4().hex
        unrelated = "mimo_unrelated_" + uuid4().hex
        alias = "mimo_archive_" + uuid4().hex
        cell = compile_sql_cell("CREATE TABLE " + name + " AS SELECT 1 AS value", "owned_cell")
        self.assertIn(name, cell.variable_data)
        with tempfile.TemporaryDirectory() as directory:
            database_path = str(Path(directory) / "archive.duckdb").replace("'", "''")
            duckdb.execute("ATTACH '" + database_path + "' AS " + alias)
            try:
                duckdb.execute("CREATE TABLE memory.main." + name + " AS SELECT 1 AS value")
                duckdb.execute("CREATE TABLE memory.main." + unrelated + " AS SELECT 2 AS value")
                duckdb.execute("CREATE TABLE " + alias + ".main." + name + " AS SELECT 99 AS value")
                self.kernel._delete_variables(cell.variable_data, set())
                self.assertEqual(memory_object_count(name), 0)
                self.assertEqual(
                    duckdb.execute("SELECT value FROM memory.main." + unrelated).fetchall(),
                    [(2,)],
                )
                self.assertEqual(
                    duckdb.execute("SELECT value FROM " + alias + ".main." + name).fetchall(),
                    [(99,)],
                )
            finally:
                duckdb.execute("DROP TABLE IF EXISTS memory.main." + name)
                duckdb.execute("DROP TABLE IF EXISTS memory.main." + unrelated)
                duckdb.execute("DETACH DATABASE IF EXISTS " + alias)

    def test_view_definitions_are_removed_from_real_duckdb(self):
        name = "mimo_view_" + uuid4().hex
        sql = "CREATE VIEW " + name + " AS SELECT e'new\\nline' AS value"
        cell = compile_sql_cell(sql, "view_cell")
        self.assertIn(name, cell.variable_data)
        try:
            duckdb.execute(sql)
            self.assertEqual(
                duckdb.execute("SELECT value FROM memory.main." + name).fetchall(),
                [("new\nline",)],
            )
            self.kernel._delete_variables(cell.variable_data, set())
            self.assertEqual(memory_object_count(name), 0)
        finally:
            duckdb.execute("DROP VIEW IF EXISTS memory.main." + name)

    def test_attached_database_definitions_are_detached(self):
        alias = "mimo_attached_" + uuid4().hex
        with tempfile.TemporaryDirectory() as directory:
            database_path = str(Path(directory) / "attached.duckdb").replace("'", "''")
            sql = "ATTACH '" + database_path + "' AS " + alias
            cell = compile_sql_cell(sql, "attach_cell")
            self.assertIn(alias, cell.variable_data)
            try:
                duckdb.execute(sql)
                self.assertIn(alias, [row[1] for row in duckdb.execute("PRAGMA database_list").fetchall()])
                self.kernel._delete_variables(cell.variable_data, set())
                self.assertNotIn(alias, [row[1] for row in duckdb.execute("PRAGMA database_list").fetchall()])
            finally:
                duckdb.execute("DETACH DATABASE IF EXISTS " + alias)


if __name__ == "__main__":
