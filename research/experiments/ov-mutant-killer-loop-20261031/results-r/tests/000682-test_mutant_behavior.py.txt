import sys
sys.path.insert(0, '/testbed')

import unittest
from unittest import mock

from c7n.resources import sns as sns_module
from c7n.resources.sns import SetEncryption


class _FakeSNSClient(object):
    def __init__(self):
        self.calls = []

    def set_topic_attributes(self, **kw):
        self.calls.append(kw)


class _FakeSession(object):
    def __init__(self, client):
        self._client = client

    def client(self, name, *a, **kw):
        assert name == 'sns'
        return self._client


class _FakeManager(object):
    def session_factory(self, *a, **kw):
        return None


class SetEncryptionBehavior(unittest.TestCase):

    resources = [
        {'TopicArn': 'arn:aws:sns:us-west-1:123456789012:one'},
        {'TopicArn': 'arn:aws:sns:us-west-1:123456789012:two'},
    ]

    def run_action(self, data):
        client = _FakeSNSClient()
        action = SetEncryption(data, _FakeManager())
        with mock.patch.object(
                sns_module, 'local_session',
                return_value=_FakeSession(client)):
            result = action.process([dict(r) for r in self.resources])
        return client, result

    def assert_all_set(self, client, value):
        self.assertEqual(len(client.calls), len(self.resources))
        self.assertEqual(
            [c['TopicArn'] for c in client.calls],
            [r['TopicArn'] for r in self.resources])
        for c in client.calls:
            self.assertEqual(c['AttributeName'], 'KmsMasterKeyId')
            self.assertEqual(c['AttributeValue'], value)

    def test_default_enables_with_aws_sns_key(self):
        client, result = self.run_action({'type': 'set-encryption'})
        self.assert_all_set(client, 'alias/aws/sns')
        self.assertEqual(len(result), 2)

    def test_explicit_enabled_true(self):
        client, _ = self.run_action({'type': 'set-encryption', 'enabled': True})
        self.assert_all_set(client, 'alias/aws/sns')

    def test_custom_key(self):
        client, _ = self.run_action(
            {'type': 'set-encryption', 'key': 'alias/my/cmk'})
        self.assert_all_set(client, 'alias/my/cmk')

    def test_disable_clears_key(self):
        client, _ = self.run_action(
            {'type': 'set-encryption', 'enabled': False, 'key': 'alias/my/cmk'})
        self.assert_all_set(client, '')


if __name__ == '__main__':
    unittest.main()
