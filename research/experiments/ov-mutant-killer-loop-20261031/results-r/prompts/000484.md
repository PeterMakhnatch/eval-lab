    ei = (eta - mu) * norm.cdf(z) + sigma * norm.pdf(z)

import unittest
import numpy as np
from scipy.stats import norm

from robo.acquisition.LogEI import LogEI


class StubModel(object):
    """Minimal deterministic model: mean(x) = x, constant predictive variance."""

    def __init__(self, var):
        self.var = var

    def predict(self, X, **kwargs):
        X = np.asarray(X, dtype=float)
        m = X[:, :1].copy()
        v = np.ones(m.shape) * self.var
        return m, v


def make_logei(var, eta):
    model = StubModel(var)
    X_lower = np.array([-10.0])
    X_upper = np.array([10.0])

    def inc(m):
        return np.array([eta])

    return LogEI(model, X_lower, X_upper, compute_incumbent=inc, par=0.0)


def evaluate(var, eta, mu):
    acq = make_logei(var, eta)
    old = np.seterr(all='ignore')
    try:
        res = acq(np.array([[mu]]))
    finally:
        np.seterr(**old)
    return float(np.ravel(res)[0])


def outcome(var, eta, mu):
    """Return ('ok', value) or ('err', None)."""
    try:
        return ('ok', evaluate(var, eta, mu))
    except Exception:
        return ('err', None)


class LogEIBehaviorTest(unittest.TestCase):

    def test_zero_variance_mean_not_better_gives_neg_inf(self):
        # sigma == 0 and predictive mean not below the incumbent: no improvement possible
        for mu in (1.0, 2.0):
            val = evaluate(0.0, 0.5, mu)
            self.assertTrue(np.isinf(val) and val < 0,
                            "expected -inf for mu=%s, got %r" % (mu, val))

    def test_mean_equal_incumbent_gives_log_sigma_pdf(self):
        # when the mean equals the incumbent, EI = sigma * pdf(0)
        for var, level in ((1.0, 1.0), (4.0, 4.0)):
            sigma = np.sqrt(var)
            val = evaluate(var, level, level)
            expected = np.log(sigma) + norm.logpdf(0.0)
            self.assertTrue(np.isfinite(val))
            self.assertAlmostEqual(val, expected, places=6)

    def test_value_is_locally_consistent_across_tiny_mean_change(self):
        # a 1e-9 change in the predictive mean must not change the code path / result
        eps = 1e-9
        for base in (1.0,):
            a = outcome(4.0, 0.5, base)
            b = outcome(4.0, 0.5, base + eps)
            self.assertEqual(a[0], b[0],
                             "inconsistent outcome kinds %r vs %r" % (a, b))
            if a[0] == 'ok':
                va, vb = a[1], b[1]
                if np.isnan(va) or np.isnan(vb):
                    self.assertTrue(np.isnan(va) and np.isnan(vb))
                elif np.isinf(va) or np.isinf(vb):
                    self.assertEqual(va, vb)
                else:
                    self.assertAlmostEqual(va, vb, places=5)


if __name__ == '__main__':
    unittest.main()
