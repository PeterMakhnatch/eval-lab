#!/usr/bin/env python3
"""Default fit bounds: only the CPE exponent is capped at 1."""

import os
import sys
import unittest

os.environ.setdefault("MPLBACKEND", "Agg")
sys.path.insert(0, "/testbed")

import numpy as np
from impedance.models.circuits import CustomCircuit


class DefaultBoundsTests(unittest.TestCase):
    def test_gerischer_time_constant_not_capped_at_one(self):
        frequencies = np.logspace(-2, 3, 60)
        r_g, t_g = 5.0, 3.0
        omega = 2 * np.pi * frequencies
        # Independent Gerischer formula plus a fixed series resistor of 1 Ohm
        impedance = 1.0 + r_g / np.sqrt(1 + 1j * omega * t_g)
        model = CustomCircuit(
            circuit="R0-G1",
            initial_guess=[4.0, 2.0],
            constants={"R0": 1.0},
        )
        model.fit(frequencies, impedance)
        np.testing.assert_allclose(model.parameters_, [r_g, t_g], rtol=1e-2)
        self.assertGreater(model.parameters_[1], 1.5)

    def test_cpe_exponent_is_capped_at_one_by_default(self):
        frequencies = np.logspace(0, 3, 40)
        omega = 2 * np.pi * frequencies
        # Data generated with an unphysical exponent of 1.3
        impedance = 1.0 + 1.0 / (1e-3 * (1j * omega) ** 1.3)
        model = CustomCircuit(
            circuit="R0-E1",
            initial_guess=[1e-3, 0.8],
            constants={"R0": 1.0},
        )
        model.fit(frequencies, impedance)
        self.assertLessEqual(model.parameters_[1], 1.0 + 1e-9)
        self.assertGreaterEqual(model.parameters_[1], 0.0)


if __name__ == "__main__":
    unittest.main()
