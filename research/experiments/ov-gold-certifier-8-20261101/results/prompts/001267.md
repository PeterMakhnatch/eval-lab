class QuerySearchHTTPBehavior(BaseTestCase):
    def response_items(self, client, term):
        response = client.get("/api/queries/search", query_string={"q": term})
        self.assertEqual(response.status_code, 200, response.data)
        items = json.loads(response.data)
        self.assertIsInstance(items, list)
        for item in items:
            self.assertIsInstance(item, dict)
            self.assertIn("id", item)
        return items

    def test_http_search_includes_name_and_description_matches_only(self):
        by_name = query_factory.create(
            name="Monthly saffronneedle revenue",
            description="Totals by region",
        )
        by_description = query_factory.create(
            name="Cohort overview",
            description="Revenue from saffronneedle orders",
        )
        unrelated = query_factory.create(
            name="Retention report",
            description="Weekly customer activity",
        )
        with app.test_client() as client, authenticated_user(client):
            items = self.response_items(client, "saffronneedle")
        ids = set(item["id"] for item in items)
        self.assertEqual(ids, set([by_name.id, by_description.id]))
        self.assertEqual(len(items), 2)
        self.assertNotIn(unrelated.id, ids)
        by_id = dict((item["id"], item) for item in items)
        self.assertEqual(by_id[by_name.id]["name"], by_name.name)
        self.assertEqual(by_id[by_description.id]["description"], by_description.description)

    def test_each_http_request_uses_its_own_search_term(self):
        invoices = query_factory.create(
            name="Warehouse invoicepine totals",
            description="Invoice amounts",
        )
        shipping = query_factory.create(
            name="Shipping overview",
            description="Delivery cedar delays by region",

        query_factory.create(name="Unrelated report", description="Other data")

            for term, expected in (
                ("invoicepine", invoices),
                ("cedar delays", shipping),
                ("invoicepine", invoices),
            ):
                items = self.response_items(client, term)
                self.assertEqual([item["id"] for item in items], [expected.id])
            self.assertEqual(
                self.response_items(client, "unmatchedzebraneedle"), []
            )
