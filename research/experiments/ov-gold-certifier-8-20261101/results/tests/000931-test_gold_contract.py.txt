import io
import sys
import unittest

sys.path.insert(0, '/testbed')

import fastavro

try:
    from fastavro._read_common import SchemaResolutionError
except ImportError:
    try:
        from fastavro.read import SchemaResolutionError
    except ImportError:
        from fastavro.schema import SchemaResolutionError


class TestSchemaResolutionComplexTypesContract(unittest.TestCase):
    """
    Contract test suite verifying schema resolution when writer or reader
    types are dicts (records, arrays, maps, enums) or dict-form primitives.
    """

    def _write_and_read(self, writer_schema, reader_schema, records):
        buf = io.BytesIO()
        fastavro.writer(buf, writer_schema, records)
        buf.seek(0)
        return list(fastavro.reader(buf, reader_schema=reader_schema))

    def test_nested_record_evolution(self):
        """Record dict schemas match when reader adds a field with default."""
        writer_schema = {
            "type": "record",
            "name": "Outer",
            "fields": [{
                "name": "nested",
                "type": {
                    "type": "record",
                    "name": "Inner",
                    "fields": [{"name": "f1", "type": "string"}],
                },
            }],
        }
        reader_schema = {
            "type": "record",
            "name": "Outer",
            "fields": [{
                "name": "nested",
                "type": {
                    "type": "record",
                    "name": "Inner",
                    "fields": [
                        {"name": "f1", "type": "string"},
                        {"name": "f2", "type": "int", "default": 42},
                    ],
                },
            }],
        }
        records = [{"nested": {"f1": "hello"}}]
        result = self._write_and_read(writer_schema, reader_schema, records)
        self.assertEqual(result, [{"nested": {"f1": "hello", "f2": 42}}])

    def test_array_of_records_evolution(self):
        """Array item record dict schemas match when evolved compatibly."""
        writer_schema = {
            "type": "record",
            "name": "Data",
            "fields": [{
                "name": "items",
                "type": {
                    "type": "array",
                    "items": {
                        "type": "record",
                        "name": "Sub",
                        "fields": [{"name": "s", "type": "string"}],
                    },
                },
            }],
        }
        reader_schema = {
            "type": "record",
            "name": "Data",
            "fields": [{
                "name": "items",
                "type": {
                    "type": "array",
                    "items": {
                        "type": "record",
                        "name": "Sub",
                        "fields": [
                            {"name": "s", "type": "string"},
                            {"name": "opt", "type": "int", "default": 0},
                        ],
                    },
                },
            }],
        }
        records = [{"items": [{"s": "foo"}]}]
        result = self._write_and_read(writer_schema, reader_schema, records)
        self.assertEqual(result, [{"items": [{"s": "foo", "opt": 0}]}])

    def test_map_of_records_evolution(self):
        """Map value record dict schemas match when evolved compatibly."""
        writer_schema = {
            "type": "record",
            "name": "MapRec",
            "fields": [{
                "name": "mapping",
                "type": {
                    "type": "map",
                    "values": {
                        "type": "record",
                        "name": "Val",
                        "fields": [{"name": "count", "type": "int"}],
                    },
                },
            }],
        }
        reader_schema = {
            "type": "record",
            "name": "MapRec",
            "fields": [{
                "name": "mapping",
                "type": {
                    "type": "map",
                    "values": {
                        "type": "record",
                        "name": "Val",
                        "fields": [
                            {"name": "count", "type": "int"},
                            {"name": "label", "type": "string", "default": "n/a"},
                        ],
                    },
                },
            }],
        }
        records = [{"mapping": {"k": {"count": 1}}}]
        result = self._write_and_read(writer_schema, reader_schema, records)
        self.assertEqual(result, [{"mapping": {"k": {"count": 1, "label": "n/a"}}}])

    def test_primitive_dict_vs_string_compatibility(self):
        """Asymmetric guards: dict in writer XOR dict in reader matches."""
        w1 = {"type": "record", "name": "R", "fields": [{"name": "x", "type": {"type": "string"}}]}
        r1 = {"type": "record", "name": "R", "fields": [{"name": "x", "type": "string"}]}
        self.assertEqual(self._write_and_read(w1, r1, [{"x": "hello"}]), [{"x": "hello"}])

        w2 = {"type": "record", "name": "R", "fields": [{"name": "x", "type": "string"}]}
        r2 = {"type": "record", "name": "R", "fields": [{"name": "x", "type": {"type": "string"}}]}
        self.assertEqual(self._write_and_read(w2, r2, [{"x": "world"}]), [{"x": "world"}])

    def test_incompatible_dict_schemas_raise(self):
        """Dict schemas that are incompatible raise SchemaResolutionError."""
        w = {
            "type": "record",
            "name": "R",
            "fields": [{
                "name": "sub",
                "type": {
                    "type": "record",
                    "name": "S",
                    "fields": [{"name": "a", "type": "int"}],
                },
            }],
        }
        r_missing_default = {
            "type": "record",
            "name": "R",
            "fields": [{
                "name": "sub",
                "type": {
                    "type": "record",
                    "name": "S",
                    "fields": [
                        {"name": "a", "type": "int"},
                        {"name": "b", "type": "string"},
                    ],
                },
            }],
        }
        with self.assertRaises(SchemaResolutionError):
            self._write_and_read(w, r_missing_default, [{"sub": {"a": 1}}])

        r_incompat_type = {
            "type": "record",
            "name": "R",
            "fields": [{
                "name": "sub",
                "type": {
                    "type": "record",
                    "name": "S",
                    "fields": [{"name": "a", "type": "string"}],
                },
            }],
        }
        with self.assertRaises(SchemaResolutionError):
            self._write_and_read(w, r_incompat_type, [{"sub": {"a": 1}}])

    def test_writer_union_containing_record_read_by_record(self):
        """Writer union with record resolved against evolved reader record."""
        writer_schema = {
            "type": "record",
            "name": "Container",
            "fields": [{
                "name": "payload",
                "type": [
                    "null",
                    {
                        "type": "record",
                        "name": "Payload",
                        "fields": [{"name": "val", "type": "int"}],
                    },
                ],
                "default": None,
            }],
        }
        reader_schema = {
            "type": "record",
            "name": "Container",
            "fields": [{
                "name": "payload",
                "type": {
                    "type": "record",
                    "name": "Payload",
                    "fields": [
                        {"name": "val", "type": "int"},
                        {"name": "extra", "type": "string", "default": "opt"},
                    ],
                },
            }],
        }
        records = [{"payload": {"val": 88}}]
        res = self._write_and_read(writer_schema, reader_schema, records)
        self.assertEqual(res, [{"payload": {"val": 88, "extra": "opt"}}])

    def test_enum_evolution_in_record_field(self):
        """Enum dict in record field matches when reader symbols are superset."""
        writer_schema = {
            "type": "record",
            "name": "EnumRec",
            "fields": [{
                "name": "status",
                "type": {
                    "type": "enum",
                    "name": "Status",
                    "symbols": ["PENDING", "ACTIVE"],
                },
            }],
        }
        reader_schema = {
            "type": "record",
            "name": "EnumRec",
            "fields": [{
                "name": "status",
                "type": {
                    "type": "enum",
                    "name": "Status",
                    "symbols": ["PENDING", "ACTIVE", "INACTIVE"],
                },
            }],
        }
        records = [{"status": "ACTIVE"}]
        res = self._write_and_read(writer_schema, reader_schema, records)
        self.assertEqual(res, [{"status": "ACTIVE"}])

    def test_schemaless_reader_with_evolved_nested_record(self):
        """Schemaless reader resolves evolved nested record dict schemas."""
        writer_schema = {
            "type": "record",
            "name": "Item",
            "fields": [{
                "name": "sub",
                "type": {
                    "type": "record",
                    "name": "S",
                    "fields": [{"name": "id", "type": "int"}],
                },
            }],
        }
        reader_schema = {
            "type": "record",
            "name": "Item",
            "fields": [{
                "name": "sub",
                "type": {
                    "type": "record",
                    "name": "S",
                    "fields": [
                        {"name": "id", "type": "int"},
                        {"name": "tag", "type": "string", "default": "def"},
                    ],
                },
            }],
        }
        buf = io.BytesIO()
        fastavro.schemaless_writer(buf, writer_schema, {"sub": {"id": 10}})
        buf.seek(0)
        res = fastavro.schemaless_reader(buf, writer_schema, reader_schema=reader_schema)
        self.assertEqual(res, {"sub": {"id": 10, "tag": "def"}})


if __name__ == '__main__':
    unittest.main()
